    Objetivo:
    Fazer com que um processo correto, ao testar outro processo correto, obtenha dele as informações de estado dos demais processos
    e atualize seu vetor de estados local com essas informações.

    Resumo:
    O programa simula um sistema distribuido com N processos, onde cada processo pode sofrer falhas e se recuperar delas.
    Cada processo armazena um vetor de estados local, onde o estado de cada processo pode ser desconhecido (-1), correto (0) ou falho (1).
    Cada processo testa periodicamente o próximo processo no anel (em ordem circular), com a função status() do SMPL.
    Caso o processo teste um processo falho, ele continua testando os próximos processos no anel até encontrar um
    processo correto ou testar todos os processos.
    O estado de cada processo testado é armazenado no vetor local State
    O resultado de cada teste é impresso na tela.
    Ao final dos testes o vetor de estados do processo atual é impresso na tela, indicando com * quais estados foram atualizados na rodada.
    O programa termina depois de atingir um tempo máximo de simulação.
*/

#include <stdio.h>
#include <stdlib.h>
#include "smpl.h"

// Definir os eventos: um processo, em algum instante de tempo, pode sofrer um evento

#define test 1
#define fault 2
#define recovery 3

// Definir o descritor do processo
typedef struct
{
    int id;     // Identificador de facility do SMPL
    int *State; // Estado de todos os outros processos: -1 = desconhecido, 0 = correto, 1 = falho
} TipoProcesso;

TipoProcesso *processo;

void cabecalho(int N, int MaxTempoSimulac)
{
    printf("===============================================================\n");
    printf("Universidade Federal do Paraná - UFPR\n");
    printf("Programa de Pós-Graduação em Informática - PPGInf\n");
    printf("Disciplina: Sistemas Distribuídos (Prof. Elias)\n\n");
    printf("Trabalho Prático 0 - Tarefa 4\n");
    printf("- Fazer cada processo correto, ao testar outro processo correto, obter as informações\n");
    printf("  do estado dos demais processos. \n\n");
    printf("Configuração da simulação:\n");
    printf("- Número de processos:  %d\n", N);
    printf("- Tempo total de simulação: %d\n", MaxTempoSimulac);
    printf("- Processo 1, 2, 3 e 4 sofre falha no tempo 31\n");
    printf("- Processo 2, 3 e 4 se recupera da falha no tempo 61\n");
    printf("- Processo 1 se recupera da falha no tempo 91\n\n");
    printf("Observações:\n");
    printf("- O estado de um processo pode ser: -1 (desconhecido), 0 (correto) ou 1 (falho)\n");
    printf("- O (*) indica que o estado foi atualiado na rodada atual, após teste direto\n");
    printf("- O (+) indica que o estado foi atualiado na rodada atual, porém obtido indiretamente do processo testado\n");
    printf("===============================================================\n");
}

int main(int argc, char *argv[])
{
    static int N,              // numero de processos do sistema distribuido
        token,                 // indica o processo que esta executando o algoritmo distribuido
        event,                 // indica o evento que ocorreu no sistema distribuido (test, fault ou recovery)
        r,                     // variavel auxiliar para receber o resultado da requisicao de falha do processo
        i,                     // variavel auxiliar para o loop de inicializacao dos processos
        MaxTempoSimulac = 210; // tempo maximo de simulacao

    static char fa_name[5]; // variavel auxiliar para armazenar o nome da facility do processo a ser criado

    if (argc != 2)
    {
        printf("Use: %s <numero de processos>\n", argv[0]);
        exit(1);
    }

    N = atoi(argv[1]);

    smpl(0, "tempo1: Programa de simulacao de sistemas distribuidos usando SMPL");
    reset();
    stream(1);

    // printf("Simulacao de um sistema distribuido com %d processos\n", N);
    cabecalho(N, MaxTempoSimulac);

    // inicializar os N processos do sistema distribuido
    processo = (TipoProcesso *)malloc(N * sizeof(TipoProcesso));
    for (i = 0; i < N; i++)
    {
        memset(fa_name, '\0', 5);
        sprintf(fa_name, "%d", i);
        processo[i].id = facility(fa_name, 1);

        processo[i].State = (int *)malloc(N * sizeof(int)); // alocar o vetor de estado local para cada processo
        for (int j = 0; j < N; j++)
        {
            processo[i].State[j] = -1; // inicializar o estado local, onde o processo não conhece o estado dos outros processos
        }
        processo[i].State[i] = 0; // o processo conhece seu próprio estado e sabe que ele é correto
    }

    // fazer o escalonamento dos eventos iniciais
    for (i = 0; i < N; i++)
    {
        schedule(test, 30.0, i); // todos os processos de 0 até N-1 vão testar na unidade de tempo 30
    }

    schedule(fault, 31.0, 1);    // o processo 1 sofre uma falha na unidade de tempo 31
    schedule(recovery, 91.0, 1); // o processo 1 se recupera na unidade de tempo 91

    // simular falha e recuperação dos demais processos (2 a N-1)
    for (i = 2; i < N; i++)
    {
        schedule(fault, 31.0, i);
        schedule(recovery, 61.0, i);
    }

    // variavel auxiliar para indicar o estado do processo alterado na rodada
    char *estado_alterado;
    estado_alterado = (char *)malloc(N * sizeof(char));

    // loop principal do simulador
    while (time() < MaxTempoSimulac)
    {
        cause(&event, &token);
        switch (event)
        {
        case test:
            // o processo está falho, então não pode testar o próximo processo
            if (status(processo[token].id) != 0)
            {
                printf("Processo %d: FALHO, nao pode testar (tempo %4.1f)\n", token, time());
            }
            else
            {
                // o processo i testa o processo j, onde j é o próximo processo na ordem circular
                int j = (token + 1) % N;
                int testou = 0; // flag para indicar se o processo testou algum processo correto
                memset(estado_alterado, ' ', N);
                while (status(processo[token].id) == 0 && j != token)
                { // o processo i testa os outros processos, exceto ele mesmo
                    if (status(processo[j].id) == 0)
                    {
                        printf("Processo %d: TESTOU PROCESSO %d - CORRETO (tempo %4.1f)\n", token, j, time());
                        processo[token].State[j] = 0; // o processo token atualiza seu estado local para o processo j como correto
                        testou = 1;                   // o processo testou um processo correto
                        estado_alterado[j] = '*';     // indica que o estado do processo j foi alterado

                        // obtem o estado dos demais processos, exceto o próprio processo e os processos anteriormente testados

                        // Mostra o vetor de estado do vetor atual
                        printf("Obtendo informações de estados do PROCESSO %d: [ ", j);
                        for (int k = 0; k < N; k++)
                        {
                            printf("%d", processo[j].State[k]); // unknown em amarelo
                            if (k < N - 1)
                                printf(", ");
                            else
                                printf(" ]\n");
                        }

                        int k = (j + 1) % N; // o próximo processo a ser buscado é o próximo na ordem circular
                        while (k != token)   // vai parar quando chegar no próprio processo
                        {
                        // o processo token (atual) atualiza seu estado local obtendo do processo j as informações de estado para o processo k
                           if( processo[j].State[k] != -1 ) { // se o processo j tem informação sobre o processo k, o processo token atualiza seu estado local para o processo k com a informação do processo j
                                processo[token].State[k] = processo[j].State[k]; 
                                estado_alterado[k] = '+'; // indica que o estado do processo k foi alterado
                            }
                            k = (k + 1) % N;                                 // o próximo processo a ser consultado é o próximo na ordem circular
                        }
                        break;
                    }
                    else
                    {
                        printf("Processo %d: TESTOU PROCESSO %d - FALHO (tempo %4.1f), continuando...\n", token, j, time());
                        processo[token].State[j] = 1; // o processo token atualiza seu estado local para o processo j como falho
                        estado_alterado[j] = '*';     // indica que o estado do processo j foi alterado
                    }
                    j = (j + 1) % N; // o próximo processo a ser testado é o próximo na ordem circular
                }
                if (!testou)
                { // o processo não testou nenhum processo correto
                    printf("Processo %d: TESTOU TODOS OS PROCESSOS - FALHO (tempo %4.1f)\n", token, time());
                }

                // Mostra o vetor de estado do processo atual
                printf("Vetor de estados PROCESSO %d: [ ", token);
                for (j = 0; j < N; j++)
                {
                    printf("%d%c", processo[token].State[j], estado_alterado[j]); // unknown em amarelo
                    if (j < N - 1)
                        printf(", ");
                    else
                        printf(" ]\n");
                }
                printf("---------------------------------------------------------------\n");
            }
            schedule(test, 30.0, token); // o processo volta a testar depois de 30 unidades de tempo
            break;

        case fault:
            r = request(processo[token].id, token, 0); // o processo token sofre uma falha
            printf("Processo %d: SOFREU FALHA (tempo %4.1f)\n", token, time());
            break;

        case recovery:
            release(processo[token].id, token); // o processo token se recupera da falha
            printf("Processo %d: RECUPERADO DA FALHA (tempo %4.1f)\n", token, time());
            break;
        }
    }

    return 0;
}
